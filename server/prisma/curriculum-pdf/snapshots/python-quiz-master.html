def ask_question(number, text, options, correct):
    """Ask one question; return True if right."""
    print(f"Question {number}: {text}")
    for index, option in enumerate(options, start=1):
        print(f"  {index}. {option}")
    while True:
        answer = input("Your answer (1-4): ")
        if answer.isdigit() and 1 <= int(answer) <= 4:
            return int(answer) == correct

def rank_for(percent):
    if percent >= 90: return "Commander"
    elif percent >= 70: return "Pilot"
    return "Cadet"
